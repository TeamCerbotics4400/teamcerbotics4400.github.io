    function getRobotMetrics(robot, filterPhase) {
      let green = 0, purple = 0, missed = 0, gate = 0;
      let parking = robot.endgame.park;

      const phases = filterPhase === 'ALL' ? ['auto', 'teleop', 'endgame'] : [filterPhase.toLowerCase()];
      phases.forEach(p => {
        if(robot[p]) {
          green += robot[p].green;
          purple += robot[p].purple;
          missed += robot[p].missed;
          gate += robot[p].gate;
        }
      });

      const scored = green + purple;
      const totalShots = scored + missed;
      const accuracy = totalShots > 0 ? ((scored / totalShots) * 100).toFixed(1) : '0.0';

      return { green, purple, scored, missed, totalShots, accuracy, gate, parking };
    }

    function drawFieldCanvas() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // Draw FTC Field Floor Grid
      ctx.strokeStyle = 'rgba(255,255,255,0.08)';
      ctx.lineWidth = 1;
      for(let x=0; x<canvas.width; x+=50) {
        ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,canvas.height); ctx.stroke();
      }
      for(let y=0; y<canvas.height; y+=50) {
        ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(canvas.width,y); ctx.stroke();
      }

      // Draw Goals
      ctx.fillStyle = 'rgba(239,68,68,0.2)'; ctx.fillRect(20, 20, 80, 80);
      ctx.fillStyle = 'rgba(59,130,246,0.2)'; ctx.fillRect(canvas.width - 100, canvas.height - 100, 80, 80);

      if(viewMode === 'LIVE') {
        // Draw Animated Tracking Vectors & Robots
        const robotsPos = [
          { x: 120 + Math.sin(currentTime)*30, y: 100 + Math.cos(currentTime)*20, color: '#ef4444', label: '11115 [RED 1]' },
          { x: 180 + Math.cos(currentTime)*40, y: 280 + Math.sin(currentTime)*30, color: '#f87171', label: '14320 [RED 2]' },
          { x: 620 - Math.sin(currentTime)*35, y: 150 + Math.cos(currentTime)*25, color: '#3b82f6', label: '8417 [BLUE 1]' },
          { x: 550 - Math.cos(currentTime)*30, y: 320 - Math.sin(currentTime)*20, color: '#60a5fa', label: '18219 [BLUE 2]' }
        ];

        robotsPos.forEach(r => {
          ctx.strokeStyle = r.color; ctx.lineWidth = 2;
          ctx.strokeRect(r.x, r.y, 50, 50);
          ctx.fillStyle = r.color; ctx.font = 'bold 10px sans-serif';
          ctx.fillText(r.label, r.x, r.y - 6);
          ctx.beginPath(); ctx.moveTo(r.x + 25, r.y + 25);
          ctx.lineTo(r.x + 25 + Math.sin(currentTime*2)*40, r.y + 25 + Math.cos(currentTime*2)*40);
          ctx.strokeStyle = 'rgba(255,255,255,0.4)'; ctx.stroke();
        });

        if(isPlaying) {
          ctx.fillStyle = '#22c55e'; ctx.beginPath();
          ctx.arc(200 + (currentTime*15)%400, 120 + Math.sin(currentTime*5)*30, 6, 0, Math.PI*2); ctx.fill();
          ctx.fillStyle = '#a855f7'; ctx.beginPath();
          ctx.arc(580 - (currentTime*12)%350, 220 + Math.cos(currentTime*4)*20, 6, 0, Math.PI*2); ctx.fill();
        }
      } else {
        // Render 2D Shot Density Heatmap Overlay

            grad.addColorStop(0.5, 'rgba(34, 197, 94, 0.3)');
            grad.addColorStop(1, 'rgba(34, 197, 94, 0)');
          } else {
            grad.addColorStop(0, 'rgba(239, 68, 68, 0.8)');
            grad.addColorStop(0.5, 'rgba(239, 68, 68, 0.3)');
            grad.addColorStop(1, 'rgba(239, 68, 68, 0)');
          }
          ctx.fillStyle = grad;
          ctx.beginPath();
          ctx.arc(pt.x, pt.y, 35, 0, Math.PI * 2);
          ctx.fill();

          // Dot Center
          ctx.fillStyle = pt.scored ? '#22c55e' : '#ef4444';
          ctx.beginPath(); ctx.arc(pt.x, pt.y, 3, 0, Math.PI * 2); ctx.fill();
        });

        // Heatmap Legend
        ctx.fillStyle = 'rgba(15, 23, 42, 0.85)';
        ctx.fillRect(15, canvas.height - 40, 240, 30);
        ctx.strokeStyle = 'rgba(255,255,255,0.1)'; ctx.strokeRect(15, canvas.height - 40, 240, 30);

    function exportCSV() {
      let csvContent = "data:text/csv;charset=utf-8,";
      csvContent += "Equipo,Nombre,Alianza,Fase,Disparos Totales,Anotados,Fallados,Precision (%),Pelotas Verdes,Pelotas Moradas,Aperturas Gate,Estacionamiento\n";

      matchData.m1.robots.forEach(r => {
        ['ALL', 'AUTO', 'TELEOP', 'ENDGAME'].forEach(phase => {
          const m = getRobotMetrics(r, phase);
          csvContent += `${r.team},"${r.name}",${r.alliance},${phase},${m.totalShots},${m.scored},${m.missed},${m.accuracy},${m.green},${m.purple},${m.gate},${m.parking}\n`;
        });
      });

      const encodedUri = encodeURI(csvContent);
      const link = document.createElement("a");
      link.setAttribute("href", encodedUri);
      link.setAttribute("download", `FTC2026_Match12_Scouting_Report.csv`);
      document.body.appendChild(link);
      link.click();
      link.remove();
    }

    function togglePlay() {
      isPlaying = !isPlaying;
      const playIcon = document.getElementById('playIcon');
      if(isPlaying) {
        playIcon.className = 'fa-solid fa-pause';
        timerInterval = setInterval(() => {
          currentTime = (currentTime + 0.5) % 150;
          document.getElementById('timeSlider').value = currentTime;
          updatePlaybackTime();
        }, 100);
      } else {
        playIcon.className = 'fa-solid fa-play';
        clearInterval(timerInterval);
      }
    }

    function seekTime(val) {
      currentTime = parseFloat(val);
      updatePlaybackTime();
    }

    function updatePlaybackTime() {
      const mins = Math.floor(currentTime / 60).toString().padStart(2, '0');
      const secs = Math.floor(currentTime % 60).toString().padStart(2, '0');
      document.getElementById('timeCurrent').innerText = `${mins}:${secs}`;

      drawFieldCanvas();
    }

    function setPhaseFilter(phase) {
      activePhaseFilter = phase;
      ['ALL', 'AUTO', 'TELEOP', 'ENDGAME'].forEach(p => {
        const btn = document.getElementById(`btnFilter${p}`);
        btn.className = (p === phase) 
          ? 'px-3 py-1.5 rounded-lg text-xs font-semibold bg-blue-600 text-white transition' 
          : 'px-3 py-1.5 rounded-lg text-xs font-semibold bg-slate-800 text-slate-400 hover:text-white transition';
      });
      updateUI();
    }
